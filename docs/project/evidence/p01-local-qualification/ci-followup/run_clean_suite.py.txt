from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys

ROOT = Path('$WORKSPACE/AI-GPU')
NOTES = ROOT.parent / 'work-notes/blockers'
sys.path.insert(0, str(ROOT / 'ci/project'))
from baseline_process import GIB, MIB, run_bounded

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, text=True, timeout=20).strip()

def now():
    return datetime.now(timezone.utc).isoformat()

def hashes():
    files = sorted((ROOT / 'ci/project').glob('*.py'))
    files.append(ROOT / 'tests/regression/stencil3d/main.cpp')
    return {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in files}

expected = sys.argv[1]
label = sys.argv[2]
if not re.fullmatch(r'[a-zA-Z0-9-]+', label):
    raise SystemExit('Invalid label')
record = NOTES / (label + '.json')
log = NOTES / (label + '.log')
before = {'source_revision': git('rev-parse', 'HEAD'), 'source_tree': git('rev-parse', 'HEAD^{tree}'),
          'status_before': git('status', '--porcelain'), 'files_before': hashes(), 'started_at': now()}
if before['source_revision'] != expected or before['status_before'] or record.exists() or log.exists():
    raise SystemExit('Requires expected clean source and new record/log')
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1')
command = [sys.executable, '-B', '-m', 'unittest', 'discover', '-s', 'ci/project', '-p', 'test_*.py']
before.update(command=command, state='running')
record.write_text(json.dumps(before, indent=2) + '\n')
result = run_bounded(command, ROOT, env, log, timeout_seconds=900, min_free_bytes=15*GIB,
                     max_rss_bytes=GIB, max_log_bytes=4*MIB, resource_paths=(NOTES,))
contents = log.read_text()
match = re.search(r'^Ran (\d+) tests? in ([\d.]+)s$', contents, re.M)
after = {'files_after': hashes(), 'status_after': git('status', '--porcelain'),
         'revision_after': git('rev-parse', 'HEAD'), 'finished_at': now()}
summary = [line for line in contents.splitlines() if line.startswith(('Ran ', 'OK', 'FAILED', 'FAIL:', 'ERROR:'))]
data = {**before, **after, 'state': 'finished', 'result': result, 'result_lines': summary,
        'tests_run': int(match.group(1)) if match else None,
        'source_files_unchanged': before['files_before'] == after['files_after']}
data['outcome'] = 'pass' if (result['outcome'] == 'pass' and match and data['source_files_unchanged']
                            and not after['status_after'] and after['revision_after'] == expected) else 'fail'
record.write_text(json.dumps(data, indent=2) + '\n')
print(json.dumps({k: data[k] for k in ('outcome', 'tests_run', 'source_files_unchanged', 'result_lines')}))
sys.exit(0 if data['outcome'] == 'pass' else 1)
