def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def close_tail(p):
    tree=ast.parse(p.read_text())
    function=next(node for node in tree.body if isinstance(node,ast.FunctionDef) and node.name=='child')
    at=next(i for i,node in enumerate(function.body) if isinstance(node,ast.Expr) and isinstance(node.value,ast.Call) and isinstance(node.value.func,ast.Attribute) and isinstance(node.value.func.value,ast.Name) and node.value.func.value.id=='adapter' and node.value.func.attr=='close')
    stop=next(i for i in range(at+1,len(function.body)) if isinstance(function.body[i],ast.For))
    return ast.unparse(ast.Module(body=function.body[at+1:stop],type_ignores=[]))
