## Portable inert verification recipe

```bash
python3 - <<'PY'
from pathlib import Path
import hashlib, json, zipfile

p = Path('docs/project/evidence/p06-native-lifetime/native/b4127b19-run45130300651-attempt1')
sha = '3d4879f73c197352689d65feb56e73f537701fc1411e55bf53c92b1b41e13db3'
archive = p / 'original.zip'
assert archive.stat().st_size == 9086055
assert hashlib.sha256(archive.read_bytes()).hexdigest() == sha
manifest = json.loads((p / 'member-inventory.json').read_text())
assert manifest['archive_sha256'] == sha and manifest['archive_bytes'] == 9086055
rows = manifest['files']
index = {row['path']: row for row in rows}
assert len(rows) == len(index) == 1264
with zipfile.ZipFile(archive) as data:
    names = data.namelist()
    assert len(names) == len(set(names)) == len(index)
    assert set(names) == set(index)
    for name, identity in index.items():
        body = data.read(name)
        assert len(body) == identity['size_bytes']
        assert hashlib.sha256(body).hexdigest() == identity['sha256']
print('Verified original archive and 1264 members; nothing extracted or executed.')
PY
```

This checks byte identity and inventory completeness. It does not replace the
