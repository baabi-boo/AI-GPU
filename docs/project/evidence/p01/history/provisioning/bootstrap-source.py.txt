import hashlib
import json
import os
from pathlib import Path
import shutil
import tarfile
import time
import urllib.request

ROOT = Path('$WORKSPACE')
DEST = ROOT / 'tools-v3.0.2'
DOWNLOAD = ROOT / 'tool-downloads'
LOG = ROOT / 'baseline-runs' / 'tool-bundles.json'
REV = 'e3db4bcdf819a17f618ad9fc2472f8f7dea3192b'
COMPONENTS = {
    'riscv64-gnu-toolchain': [f'riscv64-gnu-toolchain/ubuntu/focal/riscv64-gnu-toolchain.tar.bz2.parta{x}' for x in 'abcdefghij'],
    'llvm-vortex': [f'llvm-vortex/ubuntu/focal/llvm-vortex.tar.bz2.parta{x}' for x in 'abc'],
    'verilator': [f'verilator/ubuntu/focal/verilator.tar.bz2.parta{x}' for x in 'abc'],
    'libc64': ['libc64/libc64.tar.bz2'],
    'libcrt64': ['libcrt64/libcrt64.tar.bz2'],
}

def main():
    DOWNLOAD.mkdir(exist_ok=True)
    DEST.mkdir(exist_ok=True)
    tree_url = f'https://api.github.com/repos/vortexgpgpu/vortex-toolchain-prebuilt/git/trees/{REV}?recursive=1'
    with urllib.request.urlopen(tree_url, timeout=30) as response:
        tree = json.load(response)
    if tree.get('truncated'):
        raise RuntimeError('truncated toolchain tree')
    blobs = {item['path']: item for item in tree['tree'] if item['type'] == 'blob'}
    report = {'repository': 'vortexgpgpu/vortex-toolchain-prebuilt', 'tag': 'v3.0.2', 'revision': REV, 'os_bundle': 'ubuntu/focal', 'parts': []}
    for component, paths in COMPONENTS.items():
        for path in paths:
            item = blobs[path]
            if shutil.disk_usage(ROOT).free < 15 * 1024**3:
                raise RuntimeError('disk reserve threatened')
            destination = DOWNLOAD / Path(path).name
            if not destination.exists():
                print('download', path, item['size'], flush=True)
                url = f'https://raw.githubusercontent.com/vortexgpgpu/vortex-toolchain-prebuilt/{REV}/{path}'
                temporary = destination.with_suffix(destination.suffix + '.partial')
                for attempt in range(3):
                    try:
                        with urllib.request.urlopen(url, timeout=60) as response, temporary.open('wb') as out:
                            total = 0
                            while chunk := response.read(1024 * 1024):
                                total += len(chunk)
                                if total > item['size']:
                                    raise RuntimeError('download exceeds pinned size')
                                out.write(chunk)
                        if total != item['size']:
                            raise RuntimeError('download length mismatch')
                        temporary.rename(destination)
                        break
                    except Exception:
                        if attempt == 2:
                            raise
                        time.sleep(1)
            sha1 = hashlib.sha1(f'blob {item["size"]}\0'.encode())
            sha256 = hashlib.sha256()
            with destination.open('rb') as stream:
                while chunk := stream.read(1024 * 1024):
                    sha1.update(chunk)
                    sha256.update(chunk)
            if sha1.hexdigest() != item['sha']:
                raise RuntimeError('Git blob identity mismatch: ' + path)
            report['parts'].append({'path': path, 'size_bytes': item['size'], 'git_blob_sha': item['sha'], 'sha256': sha256.hexdigest()})
            LOG.write_text(json.dumps(report, indent=2) + '\n')
        if (DEST / component).exists():
            continue
        archive = DOWNLOAD / (component + '.assembled.tar.bz2')
        with archive.open('wb') as output:
            for path in paths:
                with (DOWNLOAD / Path(path).name).open('rb') as source:
                    shutil.copyfileobj(source, output, 1024 * 1024)
        print('extract', component, flush=True)
        with tarfile.open(archive) as bundle:
            total = 0
            for member in bundle.getmembers():
                total += member.size
                if total > 10 * 1024**3:
                    raise RuntimeError('component expansion budget exceeded')
                if Path(member.name).parts[0] != component:
                    raise RuntimeError('unexpected component root')
            bundle.extractall(DEST, filter='data')
        archive.unlink()
        print('installed', component, flush=True)
    print('toolchain complete', flush=True)

if __name__ == '__main__':
    main()
