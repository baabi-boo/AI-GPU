#!/usr/bin/env python3
"""Supplemental CLI propagation checks; no archive or native execution."""
import ast
from contextlib import redirect_stdout
from copy import deepcopy
import hashlib
import io
import json
from pathlib import Path
import tempfile
import time
from types import SimpleNamespace

ROOT = Path('/workspace/scratch/dc8be1693f91/AI-GPU')
OUT = ROOT / 'docs/project/evidence/p06-qualified-warm/validation/w07_a01-claim-probe.json'
NAMES = [
    'p06_managed_preflight_contract.py', 'p06_managed_preflight_execute.py',
    'p06_qualified_warm_execute.py', 'p06_qualified_warm_replay.py',
    'p06_qualified_warm_analyze.py', 'p06_dependency_native_replay.py',
    'p06_warm_analyze.py', 'p06_warm_contract.json',
]
def identities():
    return {name: hashlib.sha256((ROOT / 'ci/project' / name).read_bytes()).hexdigest()
            for name in NAMES}
before = identities()
started = time.monotonic()
tree = ast.parse((ROOT / 'ci/project/p06_managed_preflight_contract.py').read_text())
claim_node = next(n for n in tree.body if isinstance(n, ast.Assign)
    and any(isinstance(t, ast.Name) and t.id == 'FALSE_CLAIMS' for t in n.targets))
claims = ast.literal_eval(claim_node.value)
assert len(claims) == 7

def extracted_main(name, namespace):
    path = ROOT / 'ci/project' / name
    node = next(n for n in ast.parse(path.read_text()).body
                if isinstance(n, ast.FunctionDef) and n.name == 'main')
    program = ast.fix_missing_locations(ast.Module(body=[node], type_ignores=[]))
    exec(compile(program, str(path), 'exec'), namespace)
    return namespace['main']

rows = []
with tempfile.TemporaryDirectory(prefix='p06-w07-claims-', dir=ROOT.parent / 'validation') as directory:
    directory = Path(directory)
    for status in ('satisfied', 'not_satisfied'):
        for written in (False, True):
            target = directory / (status + '.json') if written else None
            args = SimpleNamespace(archive='unused', sha256='a' * 64, source_revision='b' * 40,
                dependency_receipt_sha256='c' * 64, native_report_sha256='d' * 64,
                source_manifest_sha256='e' * 64, counter_qualification_sha256='f' * 64,
                diagnostic_archive='unused', diagnostic_archive_sha256='a' * 64,
                control_archive='unused', control_archive_sha256='b' * 64, prefix='', output=target)
            result = {'scope': 'synthetic-cli-propagation-only', 'counter_qualification_passed': True,
                'qualified_warm_passed': True, 'native_execution': 'not_run',
                'retained_code_executed': False, **dict.fromkeys(claims, False),
                'analysis': {'hypothesis_evaluation': status, 'performance_acceptance': False,
                             'phase_acceptance': False, 'acceptance_claimed': False}}
            namespace = {'argument_parser': lambda: SimpleNamespace(parse_args=lambda argv: args),
                'replay_archive': lambda *a, **k: deepcopy(result), 'json': json,
                'contract': SimpleNamespace(FALSE_CLAIMS=claims)}
            main = extracted_main('p06_qualified_warm_replay.py', namespace)
            stream = io.StringIO()
            with redirect_stdout(stream): code = main([])
            terminal = json.loads(stream.getvalue())
            body = json.loads(target.read_text()) if written else terminal
            assert code == 0
            assert body['analysis']['hypothesis_evaluation'] == status
            assert body['qualified_warm_passed'] is True
            assert all(body[name] is False for name in claims)
            assert all(body['analysis'][name] is False for name in
                       ('performance_acceptance', 'phase_acceptance', 'acceptance_claimed'))
            if written:
                assert terminal['hypothesis_evaluation'] == status
                assert terminal['outcome'] == 'pass'
                assert all(terminal[name] is False for name in claims)
            rows.append({'case': 'reader-' + status + ('-file' if written else '-stdout'),
                         'outcome': 'pass', 'exit_code': code})

    # Both actual CLI exception branches must fail without promoting claims.
    def rejected(*a, **k): raise ValueError('synthetic complete-reader rejection')
    namespace['replay_archive'] = rejected
    stream = io.StringIO()
    with redirect_stdout(stream): code = main([])
    terminal = json.loads(stream.getvalue())
    assert code == 1 and terminal['qualified_warm_passed'] is False
    assert terminal['counter_qualification_passed'] is False
    assert all(terminal[name] is False for name in claims)
    rows.append({'case': 'reader-rejection', 'outcome': 'pass', 'exit_code': code})
    namespace['replay_archive'] = lambda *a, **k: deepcopy(result)
    original = target.read_bytes()
    stream = io.StringIO()
    with redirect_stdout(stream): code = main([])
    terminal = json.loads(stream.getvalue())
    assert code == 1 and target.read_bytes() == original
    assert terminal['qualified_warm_passed'] is False
    assert all(terminal[name] is False for name in claims)
    rows.append({'case': 'reader-existing-output', 'outcome': 'pass', 'exit_code': code})

    producer_args = SimpleNamespace(output=directory)
    preflight = SimpleNamespace(base=SimpleNamespace(resolve_paths=lambda args: 'unused'),
                               manifest=SimpleNamespace(FALSE_CLAIMS=claims))
    for passed in (True, False):
        produced = {'outcome': 'pass' if passed else 'fail', 'reason': None if passed else 'synthetic failure',
                    'completed_counts': {}, 'native_preflight_passed': passed,
                    'native_lifetime_passed': passed,
                    'dependency_source_to_loaded_binary_verified': passed,
                    'counter_qualification_passed': passed, 'qualified_warm_passed': passed}
        producer_namespace = {'argument_parser': lambda: SimpleNamespace(parse_args=lambda argv: producer_args),
            'execute': lambda *a: deepcopy(produced), 'preflight': preflight, 'json': json, 'Path': Path}
        producer_main = extracted_main('p06_qualified_warm_execute.py', producer_namespace)
        stream = io.StringIO()
        with redirect_stdout(stream): code = producer_main([])
        terminal = json.loads(stream.getvalue())
        assert code == (0 if passed else 1)
        assert terminal['qualified_warm_passed'] is passed
        assert all(terminal[name] is False for name in claims)
        assert 'hypothesis_evaluation' not in terminal
        rows.append({'case': 'producer-' + str(passed), 'outcome': 'pass', 'exit_code': code})
    producer_namespace['execute'] = rejected
    stream = io.StringIO()
    with redirect_stdout(stream): code = producer_main([])
    terminal = json.loads(stream.getvalue())
    assert code == 1 and terminal['outcome'] == 'blocked'
    assert terminal['qualified_warm_passed'] is False
    assert all(terminal[name] is False for name in claims)
    rows.append({'case': 'producer-exception', 'outcome': 'pass', 'exit_code': code})

assert identities() == before
record = {'schema_version': 1, 'scope': 'AST-extracted actual CLI propagation only',
    'outcome': 'pass', 'cases': rows, 'case_count': len(rows),
    'source_sha256': before, 'source_stable': True,
    'duration_seconds': time.monotonic() - started,
    'native_execution': 'not_run', 'archive_replay': 'not_run',
    'unique_governance_test_ids_added': 0,
    'limitations': ['Only actual CLI function bodies execute. Parser, producer and complete archive reader are explicit stubs.',
        'This does not prove a native attempt or archive is valid and does not rerun the arithmetic test suite.']}
with OUT.open('x') as stream:
    json.dump(record, stream, indent=2, sort_keys=True)
    stream.write('\n')
print(json.dumps({'outcome': record['outcome'], 'cases': len(rows), 'duration_seconds': record['duration_seconds']}))
