"""Independent baseline byte comparison and exact workflow insertion check."""
import gzip, hashlib, json, os, subprocess, time
from pathlib import Path
ROOT = Path('/workspace/scratch/dc8be1693f91/AI-GPU')
BASE = 'f80b58b06e7bcf734f3190f6db478fd11f0bab01'
OUT = ROOT / 'docs/project/evidence/p06-qualified-warm/validation/w04_a06-protected-results.json'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)
def blob_id(body):
    return hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest()
def sha(body):
    return hashlib.sha256(body).hexdigest()
start=time.monotonic()
rows=[]
links=[]
for raw in git('ls-tree','-r','-z',BASE).split(b'\0'):
    if not raw: continue
    info,name=raw.split(b'\t',1)
    mode,kind,objectid=info.decode().split()
    name=name.decode()
    if kind=='commit':
        index=git('ls-files','--stage','--',name).decode().split()
        links.append({'path':name,'baseline_gitlink':objectid,'index_gitlink':index[1],
                      'unchanged':index[1]==objectid})
        continue
    path=ROOT/name
    if mode=='120000': body=os.readlink(path).encode()
    else: body=path.read_bytes()
    actual=blob_id(body)
    rows.append({'path':name,'baseline_blob':objectid,'observed_blob':actual,
                 'sha256':sha(body),'size_bytes':len(body),'unchanged':actual==objectid})
byname={r['path']:r for r in rows}

def group(label, predicate):
    selected=[r for r in rows if predicate(r['path'])]
    return {'label':label,'files':len(selected),'bytes':sum(r['size_bytes'] for r in selected),
            'changed':[r['path'] for r in selected if not r['unchanged']]}
groups=[
    group('RTL',lambda p:p.startswith('hw/rtl/')),
    group('simulator/model',lambda p:p.startswith('sim/')),
    group('native runtime/kernel/shared ABI',lambda p:p.startswith('sw/')),
    group('golden baselines',lambda p:p.startswith('ci/baselines/')),
    group('gate implementation and catalog',lambda p:p.startswith('ci/testcases/') or p in ('ci/perf_baseline.py','ci/testcase.py') or (p.startswith('ci/') and p.endswith('_gate.py'))),
    group('all existing benchmark contracts and fixtures',lambda p:p.startswith('ci/project/fixtures/') or (p.startswith('ci/project/') and ('benchmark_contract' in p or 'warm_contract' in p)) or p.startswith(('tests/regression/ai_bench_','tests/regression/ai_warm_workspace/'))),
    group('frozen native and warm experiment implementation',lambda p:p.startswith(('ci/project/p06_warm_','ci/project/p06_benchmark_','ci/project/p06_contract.','ci/project/p06_analyze.','ci/project/p06_execute.'))),
    group('all existing proof JSON',lambda p:p.startswith('ci/project/') and p.endswith('_proof.json')),
    group('toolchain/build/configuration',lambda p:p in ('VERSION','.gitmodules','configure','config.mk.in','Makefile.in','VX_config.toml','VX_types.toml','third_party/Makefile') or p.startswith(('ci/toolchain','ci/install_','profiles/'))),
    group('all historical evidence',lambda p:p.startswith('docs/project/evidence/')),
    group('roadmap/acceptance and protected README image',lambda p:p in ('docs/project/roadmap.json','docs/project/acceptance.md','README.md','docs/assets/img/vortex_framework_stack.svg')),
]
assert all(not g['changed'] for g in groups),groups
assert all(r['unchanged'] for r in links),links
