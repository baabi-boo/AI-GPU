def restore(text, changes):
    for newer, older in changes:
        assert text.count(newer) == 1, newer
        text = text.replace(newer, older)
    return text


class Case(unittest.TestCase):
    def __init__(self, index):
        super().__init__('test_fixture')
        self.index = index

    def id(self):
        return 'diagnostic.fixture_' + str(self.index)

    def setUp(self):
        self.fixture = bytearray(4096)

    def test_fixture(self):
        self.assertEqual(len(self.fixture), 4096)


def exact_statement_probe(worker_text, corrected):
    tree = ast.parse(worker_text)
    run = next(node for node in ast.walk(tree)
               if isinstance(node, ast.Assign) and isinstance(node.value, ast.Call)
               and ast.unparse(node.value).startswith('unittest.TextTestRunner('))
    selected_nodes = [run]
    if corrected:
        selected_nodes = sorted((node for node in ast.walk(tree)
                                 if isinstance(node, (ast.Assign, ast.Delete)) and
                                 run.lineno - 2 <= node.lineno <= run.lineno), key=lambda node: node.lineno)
        assert len(selected_nodes) == 3
    started, completed = [], []

    class RecordedResult(unittest.TextTestResult):
        def startTest(self, test):
            started.append(test.id())
            super().startTest(test)

        def stopTest(self, test):
            completed.append(test.id())
            super().stopTest(test)

    environment = {'unittest': unittest, 'sys': types.SimpleNamespace(stdout=io.StringIO()),
                   'RecordedResult': RecordedResult, 'all_cases': [Case(index) for index in range(4)]}
    environment['cases'] = list(environment['all_cases'])
    selected = [case.id() for case in environment['cases']]
    references = [weakref.ref(case) for case in environment['cases']]
    program = ast.fix_missing_locations(ast.Module(body=selected_nodes, type_ignores=[]))
    exec(compile(program, '<exact reviewed runner statements>', 'exec'), environment)
    gc.collect()
    result = environment['result']
    live_count = sum(reference() is not None for reference in references)
    assert result.testsRun == 4 and result.wasSuccessful()
    assert started == completed == selected and len(set(selected)) == 4
    assert all(not getattr(result, name) for name in
               ('failures', 'errors', 'skipped', 'expectedFailures', 'unexpectedSuccesses'))
    assert live_count == (0 if corrected else 4)
    if corrected:
        assert environment['suite']._tests == [None] * 4
        assert 'all_cases' not in environment and 'cases' not in environment
    return {'corrected': corrected, 'synthetic_cases_run': result.testsRun,
            'unique_synthetic_ids': len(set(selected)), 'live_completed_instances': live_count,
            'started_equals_completed_equals_selected': True, 'adverse_outcomes': 0,
            'executed_source': ast.unparse(program)}
