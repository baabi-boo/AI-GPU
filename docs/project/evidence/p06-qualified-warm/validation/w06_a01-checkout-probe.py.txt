def git(directory,*args):
    return subprocess.check_output(['git','-C',str(directory),*args])
def textgit(directory,*args): return git(directory,*args).decode().strip()
def sha256(path):
    with path.open('rb') as stream: return hashlib.file_digest(stream,'sha256').hexdigest()
def signature(st): return (st.st_dev,st.st_ino,st.st_size,st.st_mtime_ns,st.st_ctime_ns,st.st_nlink)
def canonical(path,existing=True):
    assert path.is_absolute() and path.resolve(strict=existing)==path
    assert all(not x.is_symlink() for x in (path,*path.parents))

def inspect_repo(directory,relative,expected):
    assert textgit(directory,'rev-parse','HEAD')==expected
    assert not git(directory,'status','--porcelain=v1','--untracked-files=all','--ignored')
    rows=git(directory,'ls-tree','-r','-z','HEAD').split(b'\0')[:-1]
    current=[]
    for raw in rows:
        meta,pathraw=raw.split(b'\t',1); mode,kind,oid=meta.decode().split(); name=pathraw.decode()
        target=directory/name; display=(relative+'/'+name).lstrip('/')
        if kind=='commit':
            gitlinks.append({'path':display,'revision':oid,'mode':mode})
            inspect_repo(target,display,oid); continue
        assert kind=='blob'
        before=target.lstat()
        if mode=='120000':
            assert stat.S_ISLNK(before.st_mode)
            body=os.fsencode(os.readlink(target)); h=hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest(); size=len(body); sh=hashlib.sha256(body).hexdigest()
        else:
            assert stat.S_ISREG(before.st_mode)
            size=before.st_size; git_hash=hashlib.sha1(b'blob '+str(size).encode()+b'\0'); digest=hashlib.sha256()
            with target.open('rb') as stream:
                while chunk:=stream.read(1024*1024): git_hash.update(chunk); digest.update(chunk)
            h=git_hash.hexdigest(); sh=digest.hexdigest()
            assert bool(before.st_mode & stat.S_IXUSR)==(mode=='100755')
        assert signature(before)==signature(target.lstat())
        assert h==oid, display
        row={'path':display,'mode':mode,'git_blob':oid,'sha256':sh,'size_bytes':size}
        inventory.append(row); current.append(row)
    assert textgit(directory,'rev-parse','HEAD')==expected
    assert not git(directory,'status','--porcelain=v1','--untracked-files=all','--ignored')
    repositories.append({'path':relative or '.', 'revision':expected,'tree':textgit(directory,'rev-parse','HEAD^{tree}'),'tracked_blobs':len(current),'tracked_bytes':sum(x['size_bytes'] for x in current),'clean_including_ignored':True})
