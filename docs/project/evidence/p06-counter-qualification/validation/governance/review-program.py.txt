def sha(body):
    return hashlib.sha256(body).hexdigest()

def identity(path):
    body = path.read_bytes()
    return {'path': str(path), 'sha256': sha(body), 'size_bytes': len(body)}

def git(source, *args):
    return subprocess.check_output(['git', *args], cwd=source, text=True).strip()

def exact_source(source, revision, tree, expected_rows):
    assert git(source, 'rev-parse', 'HEAD') == revision
    assert git(source, 'rev-parse', 'HEAD^{tree}') == tree
    assert git(source, 'status', '--porcelain=v1', '--untracked-files=all') == ''
    for row in expected_rows:
        assert sha((source / row['path']).read_bytes()) == row['sha256'], row['path']
    entries = subprocess.check_output(['git', 'ls-tree', '-r', '-z', revision], cwd=source).split(b'\0')[:-1]
    blobs = []
    modules = []
    for entry in entries:
        fields, name = entry.split(b'\t', 1)
        mode, kind, oid = fields.decode().split()
        name = name.decode()
        path = source / name
        if mode == '160000':
            assert git(path, 'rev-parse', 'HEAD') == oid
            assert git(path, 'status', '--porcelain=v1', '--untracked-files=all') == ''
            modules.append({'path': name, 'revision': oid})
            continue
        assert kind == 'blob'
        if mode == '120000':
            assert path.is_symlink()
            body = os.readlink(path).encode()
        else:
            assert stat.S_ISREG(path.stat().st_mode) and not path.is_symlink()
            body = path.read_bytes()
            assert bool(path.stat().st_mode & 0o111) == (mode == '100755'), name
        actual_oid = hashlib.sha1(b'blob ' + str(len(body)).encode() + b'\0' + body).hexdigest()
        assert actual_oid == oid, name
        blobs.append({'path': name, 'mode': mode, 'git_oid': oid, 'sha256': sha(body)})
    return {'revision': revision, 'tree': tree, 'clean_status': True,
        'snapshot_rows_verified': len(expected_rows), 'git_blobs_verified': len(blobs),
        'git_blob_inventory_sha256': sha(json.dumps(blobs, sort_keys=True, separators=(',', ':')).encode()),
        'initialized_top_level_submodules': modules,
        'recursive_submodule_status': git(source, 'submodule', 'status', '--recursive')}

def store_run_files(name, directory):
    files = sorted(path for path in directory.iterdir() if path.is_file()
                   and path.suffix in ('.json', '.log', '.py'))
    preserved = directory / 'preserved-v1'
    if preserved.is_dir():
        files += sorted(path for path in preserved.iterdir() if path.is_file()
                        and path.suffix in ('.json', '.log', '.py'))
    target = OUT / name
    target.mkdir(exist_ok=True)
    rows = []
    for path in files:
        body = path.read_bytes()
        relative = path.relative_to(directory)
        if path.suffix == '.py':
            stored_name, stored, encoding = str(relative) + '.txt', body, 'identity'
        elif relative == Path('execution.json'):
            stored_name, stored, encoding = str(relative), body, 'identity'
        else:
            stored_name, stored, encoding = str(relative) + '.gz', gzip.compress(body, mtime=0), 'gzip'
        destination = target / stored_name
        destination.parent.mkdir(parents=True, exist_ok=True)
        if destination.exists():
            assert destination.read_bytes() == stored, 'retained original cannot change: ' + str(destination)
        else:
            destination.write_bytes(stored)
        assert (gzip.decompress(stored) if encoding == 'gzip' else stored) == body
        assert path.read_bytes() == body, 'input changed during retention: ' + str(path)
        rows.append({'original_path': str(path), 'original_sha256': sha(body),
            'original_size_bytes': len(body), 'retained_path': str(destination.relative_to(OUT)),
            'retained_sha256': sha(stored), 'retained_size_bytes': len(stored), 'encoding': encoding})
    return rows
